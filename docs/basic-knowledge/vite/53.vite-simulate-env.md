Completely "rebuilding a Vite's import.meta.env system" can be done entirely with plugins.

## 🚀Ultimate goal {#🚀最终目标}

With your own plugin, developers can:

```js
console.log(import.meta.env.VITE_API_URL)
```

The functionality is **exactly consistent** with Vite’s built-in behavior:

- ✔ Automatically read `.env*`
- ✔ Distinguish between `mode` (development / production)
- ✔ Only expose `VITE_` prefix variables
- ✔ Automatically inject `BASE_URL`, `MODE`, `DEV`, `PROD`
- ✔ Replace `import.meta.env` on client side
- ✔Support HMR

The overall behavior is **99%** the same as the one built into Vite.

## 🧩 Vite’s built-in import.meta.env principle {#🧩-vite-内置-import-meta-env-原理}

Vite's import.meta.env **Not a virtual module** , instead:

#### 1. Parse env files by pattern {#_1-按模式解析-env-文件}

```
.env
.env.local
.env.development
.env.development.local
...
```

#### 2. Merge out "User-Accessible Environment Variables" {#_2-合并出-用户可访问的环境变量}

Expose only:

* `VITE_` prefix variable
* `MODE`
* `DEV`
* `PROD`
* `BASE_URL`

This constitutes the final:

```ts
import.meta.env = {
  MODE: 'development',
  DEV: true,
  PROD: false,
  BASE_URL: '/',
  VITE_API_URL: '...',
  ...
}
```

#### 3. Code replacement via **Esbuild transform stage** {#_3-通过-esbuild-transform-阶段进行代码替换}

Vite scan:

```
import.meta.env.VITE_API_URL
```

Replace with **static literal string**:

```js
"some-url"
```

#### 4. Support HMR (modify env file → full page reload) {#_4-支持-hmr-修改-env-文件-→-整页-reload}

## 🏗️ 1. Large plug-in framework {#🏗️-一、插件大框架}

```js
import fs from 'node:fs'
import path from 'node:path'
import dotenv from 'dotenv'

export default function envCorePlugin() {
  let config
  let env = {}

  return {
    name: 'vite-plugin-env-core',

    configResolved(resolvedConfig) {
      config = resolvedConfig
      env = loadEnv(config)
    },

    // 1. 替换 import.meta.env
    transform(code, id) {
      if (!id.endsWith('.js') && !id.endsWith('.ts') && !id.includes('.vue')) {
        return
      }
      return transformImportMetaEnv(code, env)
    },

    // 2. HMR：监听 .env 文件变化
    handleHotUpdate(ctx) {
      if (ctx.file.includes('.env')) {
        env = loadEnv(config)
        ctx.server.ws.send({
          type: 'full-reload',
          path: '*'
        })
      }
    }
  }
}
```

Next we complete each part.

## 🧠 2. Implement loadEnv - Completely simulate Vite's env loading logic {#🧠-二、实现-loadenv-——-完整模拟-vite-的-env-加载逻辑}

Vite original logic (after simplification):

```ts
const files = [
  `.env`,
  `.env.local`,
  `.env.${mode}`,
  `.env.${mode}.local`
]
```

Let's do the same:

```js
function loadEnv(config) {
  const mode = config.mode
  const envDir = config.envDir || process.cwd()

  const files = [
    `.env`,
    `.env.local`,
    `.env.${mode}`,
    `.env.${mode}.local`
  ]

  let raw = {}

  for (const file of files) {
    const full = path.resolve(envDir, file)
    if (fs.existsSync(full)) {
      const parsed = dotenv.parse(fs.readFileSync(full))
      Object.assign(raw, parsed)
    }
  }

  // 只暴露 VITE_ 前缀变量
  const clientEnv = {}
  for (const k in raw) {
    if (k.startsWith('VITE_')) {
      clientEnv[k] = raw[k]
    }
  }

  // 内置变量
  clientEnv.MODE = mode
  clientEnv.DEV = mode === 'development'
  clientEnv.PROD = mode === 'production'
  clientEnv.BASE_URL = config.base || '/'

  return clientEnv
}
```

This completely simulates Vite's env merging rules.

## 🎯 3. Implement transformImportMetaEnv (core) {#🎯-三、实现-transformimportmetaenv-核心}

Goal: Put all code in:

```
import.meta.env.X
```

Replaced with static literals.

For example:

```js
console.log(import.meta.env.VITE_API_URL)
```

Replace with:

```js
console.log("https://example.com")
```

Implementation:

```js
function transformImportMetaEnv(code, env) {
  let replaced = code

  for (const key in env) {
    const value = JSON.stringify(env[key])
    replaced = replaced.replace(
      new RegExp(`import\\.meta\\.env\\.${key}`, 'g'),
      value
    )
  }

  return replaced
}
```

This achieves the same as Vite: **static replacement of import.meta.env**.

## 🧪 5. Complete plug-in code (can be run directly) {#🧪-五、完整插件代码-可直接运行}

```js
import fs from 'node:fs'
import path from 'node:path'
import dotenv from 'dotenv'

export default function envCorePlugin() {
  let config
  let env = {}

  function loadEnv(config) {
    const mode = config.mode
    const envDir = config.envDir || process.cwd()

    const files = [
      `.env`,
      `.env.local`,
      `.env.${mode}`,
      `.env.${mode}.local`
    ]

    let raw = {}
    for (const f of files) {
      const full = path.resolve(envDir, f)
      if (fs.existsSync(full)) {
        const parsed = dotenv.parse(fs.readFileSync(full))
        Object.assign(raw, parsed)
      }
    }

    const clientEnv = {}
    for (const key in raw) {
      if (key.startsWith('VITE_')) {
        clientEnv[key] = raw[key]
      }
    }

    clientEnv.MODE = mode
    clientEnv.DEV = mode === 'development'
    clientEnv.PROD = mode === 'production'
    clientEnv.BASE_URL = config.base || '/'

    return clientEnv
  }

  function transformImportMetaEnv(code, env) {
    let replaced = code

    for (const key in env) {
      const value = JSON.stringify(env[key])
      replaced = replaced.replace(
        new RegExp(`import\\.meta\\.env\\.${key}`, 'g'),
        value
      )
    }

    return replaced
  }

  return {
    name: 'vite-plugin-env-core',

    configResolved(resolved) {
      config = resolved
      env = loadEnv(config)
    },

    transform(code, id) {
      if (!id.match(/\.(js|ts|vue|jsx|tsx)$/)) return
      return transformImportMetaEnv(code, env)
    },

    handleHotUpdate(ctx) {
      if (ctx.file.includes('.env')) {
        env = loadEnv(config)
        ctx.server.ws.send({
          type: 'full-reload',
          path: '*'
        })
      }
    }
  }
}
```

This is **an import.meta.env plugin** that completely recreates Vite's built-in behavior.
