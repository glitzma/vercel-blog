 **Both development (dev) and build (build) phases support tree-shaking**, generate TypeScript types (`.d.ts`), and include packaging scripts, Rollup configuration and release instructions.

> target output
>
> 1. `vite-plugin-auto-all` (TypeScript implementation)
> 2. Support: `virtual:auto-api` + `virtual:env` (dev & build)
> 3. Generate **real ES module output** during the build phase (no side effects) for Rollup/webpack to do tree-shaking
> 4. Generate `.d.ts` type declaration file (rollup-plugin-dts)
> 5. Package.json that can be directly `npm publish` + release process description

## 1\. Design points {#一、设计要点}

* **Tree-shaking principle**: The exported virtual module will be built with **named exports** as pure ES modules (top-level code without side effects), and be recognized and treeshaked as Rollup chunks.So only exported functions/constants will be retained when they are used, otherwise they will be shaken off.
* **Development (dev) mode**: Provide virtual modules through `resolveId/load`, support HMR (invalidate module + return module node).
* **Build mode**: Still passes `resolveId/load`, but we **make sure the module content is a pure ES module** (only exported, no side effects are performed at the top level).Rollup treats these modules as regular modules and participates in tree-shaking.
* **TypeScript support**: The code is written in TypeScript, packaged output `dist/` (ESM & CJS), and generates type declaration `dist/index.d.ts`.
* **No side effects**: package.json `sideEffects: false` and let the output be ES modules.

## 3\. Core source code (TypeScript) {#三、核心源码-typescript}

Place the following files into `src/`.

### `src/virtualEnv.ts`

```ts
import fs from 'node:fs'
import path from 'node:path'
import dotenv from 'dotenv'

export function loadClientEnv(root: string, mode: string, prefix = 'VITE_') {
  const envDir = root || process.cwd()
  const files = [
    `.env`,
    `.env.local`,
    `.env.${mode}`,
    `.env.${mode}.local`
  ]

  const raw: Record<string,string> = {}

  for (const f of files) {
    const full = path.resolve(envDir, f)
    if (fs.existsSync(full)) {
      const parsed = dotenv.parse(fs.readFileSync(full, 'utf-8'))
      Object.assign(raw, parsed)
    }
  }

  const clientEnv: Record<string, any> = {}
  for (const k of Object.keys(raw)) {
    if (k === 'NODE_ENV' || k.startsWith(prefix)) {
      clientEnv[k] = raw[k]
    }
  }

  return clientEnv
}

/**
 * Generate pure ES module string for virtual:env
 * Exports both named exports and default export object.
 */
export function generateEnvModuleCode(env: Record<string, any>, extra = {}) {
  const merged = { ...extra, ...env }
  const keys = Object.keys(merged)

  const named = keys.map(k => `export const ${k} = ${JSON.stringify(merged[k])};`).join('\n')
  const defaultExport = `export default { ${keys.join(', ')} };`

  // No top-level side effects: only exports.
  return `${named}\n\n${defaultExport}\n`
}
```

### `src/virtualApi.ts`

```ts
import fs from 'node:fs/promises'
import path from 'node:path'

/**
 * Scan apiDir recursively and collect named exports from each file.
 * For simplicity we statically parse `export function NAME` and `export const NAME =` and `export { A, B }`
 * For production you'd use an AST parser (es-module-lexer / @babel/parser).
 */
export async function scanApiDir(root: string, apiDir: string) {
  const dir = path.resolve(root, apiDir)
  const files: string[] = []
  async function walk(dirPath: string) {
    const entries = await fs.readdir(dirPath, { withFileTypes: true })
    for (const ent of entries) {
      const p = path.join(dirPath, ent.name)
      if (ent.isDirectory()) await walk(p)
      else if (/\.(js|ts|jsx|tsx)$/.test(ent.name)) files.push(p)
    }
  }

  try {
    await walk(dir)
  } catch (e) {
    return { files: [], exports: {} }
  }

  const exportsMap: Record<string, { id: string, names: string[] }> = {}

  for (const f of files) {
    const code = (await fs.readFile(f, 'utf-8')).toString()
    const relId = '/' + path.relative(root, f).replace(/\\/g, '/')
    const names = new Set<string>()

    // simple regex-based extraction (sufficient and fast). For production, swap with AST.
    for (const m of code.matchAll(/export\s+(?:function|const|let|var|class)\s+([A-Za-z0-9_$]+)/g)) {
      names.add(m[1])
    }
    for (const m of code.matchAll(/export\s*{\s*([^}]+)\s*}/g)) {
      const parts = m[1].split(',').map(s => s.trim().split(/\s+as\s+/)[0])
      for (const p of parts) if (p) names.add(p)
    }

    if (names.size > 0) {
      exportsMap[relId] = { id: relId, names: Array.from(names) }
    }
  }

  return { files, exports: exportsMap }
}

/**
 * Produce module code that imports used functions and re-exports them as named exports.
 * This module is pure ES module and has no side effects.
 */
export function generateAutoApiModuleCode(exportsMap: Record<string, { id: string, names: string[] }>) {
  const importLines: string[] = []
  const exportNames: string[] = []

  for (const rel in exportsMap) {
    const info = exportsMap[rel]
    if (info.names.length === 0) continue
    importLines.push(`import { ${info.names.join(', ')} } from '${info.id}';`)
    exportNames.push(...info.names)
  }

  const unique = Array.from(new Set(exportNames))
  const exportLine = `export { ${unique.join(', ')} };`
  return `${importLines.join('\n')}\n\n${exportLine}\n`
}
```

### `src/index.ts` (plug-in entry) {#src-index-ts-插件入口}

```ts
import { Plugin } from 'vite'
import path from 'node:path'
import { loadClientEnv, generateEnvModuleCode } from './virtualEnv.js'
import { scanApiDir, generateAutoApiModuleCode } from './virtualApi.js'

export interface Options {
  apiDir?: string
  envPrefix?: string
}

const VIRTUAL_API_ID = 'virtual:auto-api'
const VIRTUAL_ENV_ID = 'virtual:env'
const RES_API_ID = '\0' + VIRTUAL_API_ID
const RES_ENV_ID = '\0' + VIRTUAL_ENV_ID

export default function autoAllPlugin(options: Options = {}): Plugin {
  const apiDir = options.apiDir || 'src/composables'
  const envPrefix = options.envPrefix || 'VITE_'

  let root = process.cwd()
  let mode = 'development'
  let cachedEnv: Record<string, any> = {}
  let cachedApiExports: Record<string, { id: string, names: string[] }> = {}

  return {
    name: 'vite-plugin-auto-all',
    enforce: 'pre',
    apply: 'serve', // default apply to serve; we'll still handle build with generic hooks
    configResolved(config) {
      root = config.root || root
      mode = config.mode || mode
      cachedEnv = loadClientEnv(root, mode, envPrefix)
    },

    resolveId(id) {
      if (id === VIRTUAL_API_ID) return RES_API_ID
      if (id === VIRTUAL_ENV_ID) return RES_ENV_ID
    },

    async load(id) {
      if (id === RES_ENV_ID) {
        // Always regenerate on load to pick up config
        const code = generateEnvModuleCode(cachedEnv, {
          MODE: mode,
          DEV: mode === 'development',
          PROD: mode === 'production',
          BASE_URL: '/'
        })
        return code
      }

      if (id === RES_API_ID) {
        const { exports } = await scanApiDir(root, apiDir)
        cachedApiExports = exports
        const code = generateAutoApiModuleCode(exports)
        return code
      }
    },

    // HMR handling for dev server
    async handleHotUpdate(ctx) {
      const file = ctx.file
      if (!file) return

      // if API files changed, invalidate virtual:auto-api
      const apiRoot = path.resolve(root, apiDir)
      if (file.startsWith(apiRoot)) {
        const mod = ctx.server.moduleGraph.getModuleById(RES_API_ID)
        if (mod) {
          ctx.server.moduleGraph.invalidateModule(mod)
          return [mod]
        }
      }

      // env files change -> invalidate env module and trigger full reload
      if (file.includes('.env')) {
        cachedEnv = loadClientEnv(root, mode, envPrefix)
        const mod = ctx.server.moduleGraph.getModuleById(RES_ENV_ID)
        if (mod) {
          ctx.server.moduleGraph.invalidateModule(mod)
          return [mod]
        } else {
          // fallback full reload
          ctx.server.ws.send({ type: 'full-reload' })
        }
      }
    },

    // Build-time: ensure virtual modules are provided during build as pure ES modules
    async buildStart() {
      // pre-scan API and env so Rollup sees these modules at bundle time
      cachedEnv = loadClientEnv(root, mode, envPrefix)
      const r = await scanApiDir(root, apiDir)
      cachedApiExports = r.exports
    },

    // When rollup requests the virtual id via resolve/load, we already support them above.
    // Note: apply:'serve' doesn't run for build hooks; to ensure build uses it, we can accept both:
    // (Vite will call load/resolveId during build as well)
  }
}
```

> Note: `scanApiDir` in the source code above uses regular parsing export symbols; in the production environment, it is recommended to replace it with `es-module-lexer` or `@babel/parser` to obtain more robust export recognition (especially re-export, export from, etc.).

## 4\. Packaging configuration (Rollup) {#四、打包配置-rollup}

`rollup.config.mjs` (written in ESM):

```mjs
import typescript from '@rollup/plugin-typescript'
import dts from 'rollup-plugin-dts'
import pkg from './package.json' assert { type: 'json' }

export default [
  // ESM + CJS bundle
  {
    input: 'src/index.ts',
    output: [
      { file: pkg.module, format: 'es', sourcemap: true },
      { file: pkg.main, format: 'cjs', sourcemap: true }
    ],
    external: [...Object.keys(pkg.dependencies || {}), ...Object.keys(pkg.peerDependencies || {})],
    plugins: [
      typescript({ tsconfig: './tsconfig.json', declaration: false, rootDir: 'src' })
    ]
  },
  // d.ts
  {
    input: './dist/types/src/index.d.ts',
    output: [{ file: pkg.types, format: 'es' }],
    plugins: [dts()]
  }
]
```

> Note: The above example assumes that you use `tsc` to first output the declaration to `dist/types`, and then aggregate it with rollup-plugin-dts.You can also use `tsc --emitDeclarationOnly` directly with the rollup dts step.

## 7\. Key things to note before publishing (let tree-shaking take effect) {#七、发布前的关键注意事项-让-tree-shaking-生效}

1. **Module export form**: Virtual module exports must be **named exports** or `export default` objects, and **do not perform side effects** at the top level (for example, do not perform network requests, console.log, etc. at the top of the module).The above implementation satisfies this.
2. **package.json `sideEffects: false`**: Tells the packager that the module has no side effects, allowing tree shaking.
3. **Generate ESM output**: The `module` field points to the ES module `dist/index.esm.js`, Rollup/webpack can perform tree-shaking better.
4. **Ensure virtual modules are accessible during the build phase**: Our plugin pre-scans API/Env in buildStart so that load/resolveId can be processed during the build phase; in extreme cases, more complex processing can be done in `buildStart` such as `this.emitFile({ type: 'chunk', id: '\0virtual:auto-api' })`, but in most scenarios `load` + `resolveId`Enough.

## 8\. Usage examples {#八、使用范例}

Install in target project:

```bash
npm i -D vite-plugin-auto-all
```

`vite.config.ts`：

```ts
import { defineConfig } from 'vite'
import autoAll from 'vite-plugin-auto-all'

export default defineConfig({
  plugins: [ autoAll({ apiDir: 'src/composables' }) ]
})
```

Use:

```ts
// 可以直接从虚拟模块导入
import { useUser } from 'virtual:auto-api'
import { VITE_API_URL } from 'virtual:env'

// 也可 default import env
import env from 'virtual:env'
```

In the build product, **if `useUser` is not used, this symbol will be tree-shaken** (provided that the downstream build supports tree-shaking, usually Rollup/webpack/Rspack/esbuild).

## 9\. Suggestions for generating type (`.d.ts`) {#九、生成类型-d-ts-建议}

* Export the option type of the plug-in and the related types of the virtual module (such as `declare module 'virtual:env'`) in `src/types.ts` so that TypeScript projects can recognize it.
* You can add in `dist/index.d.ts`:

```ts
declare module 'virtual:env' {
  const env: { [key: string]: string | boolean }
  export default env
  export const MODE: string
  export const DEV: boolean
  export const PROD: boolean
  export const BASE_URL: string
}
declare module 'virtual:auto-api' {
  // You may leave it open; consumers get types from actual source files.
  export {}
}
```

* `rollup-plugin-dts` can help merge `.d.ts` into `dist/index.d.ts`.

## 10\. Release process (checklist + command) {#十、发布流程-检查清单-命令}

1. Update `package.json` (name/version/author/license).Make sure package names are unique.
2. `npm run build` (or CI automation)
3. Login npm: `npm login`
4. Publish: `npm publish --access public` (if it is a scoped package, use `--access public`)
5. Tag and Release (optional): `git tag v1.0.0 && git push --tags`

CI example (GitHub Actions): trigger `npm publish` when `release` branch is tagged.

## 11\. How to locally verify that tree-shaking is effective {#十一、如何本地验证-tree-shaking-有效}

1. Install local packages (`npm pack` or `yarn add ../path/to/your/plugin`) in a test project.
2. In the test project `import { unusedFunc } from 'virtual:auto-api'` **Do not reference it in the code**, build `npm run build`.
3. Check the packaged product (such as `dist/assets/*.js`) and confirm that `unusedFunc` does not appear in the final bundle (or use source-map analysis).
4. You can also manually add `console.log('side effect')` in the plugin to confirm whether `sideEffects: false` takes effect (console.log does not exist).

## 12\. Subsequent enhancement suggestions (you can choose to implement them) {#十二、后续增强建议-你可以选择实现}

* Replace the current regular expression export detection with `es-module-lexer` or `@babel/parser` to support complex usages such as `export { default as X } from './x'`.
* Support `auto import` to generate import statements based on usage location on demand (closer to unplugin-auto-import).
* Generate `env.d.ts` for `virtual:env` and support `import.meta.env` replacement (AST + magic-string), making `import.meta.env.FOO` type-checked and auto-completed in TS.
* Provides a list of commonly used APIs built into `preset` (such as Vue preset, React preset).
* During the build phase, explicitly emit the virtual module into a chunk (`this.emitFile({ type: 'chunk', id: virtualId })`) to allow for finer rollup control (optional).

## GitHub Actions automatically publish to npm {#github-actions自动发布到npm}
You just need to put it into:

```
.github/workflows/publish.yml
```

### ✅ GitHub Actions: Automatic build + Automatic release npm + Automatic creation of Release {#✅-github-actions-自动-build-自动发布-npm-自动创建-release}

```yaml
name: Publish to npm

on:
  push:
    tags:
      - "v*" # 例如 v1.0.0 时触发

jobs:
  build-and-publish:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Use Node.js
        uses: actions/setup-node@v4
        with:
          node-version: "20"
          registry-url: "https://registry.npmjs.org"

      - name: Install dependencies
        run: npm ci

      - name: Run tests
        run: npm test --if-present

      - name: Build plugin
        run: npm run build

      # ---- 发布到 npm ----
      - name: Publish to npm
        run: npm publish --access public
        env:
          NODE_AUTH_TOKEN: ${{ secrets.NPM_TOKEN }}

      # ---- 自动创建 GitHub Release ----
      - name: Create GitHub Release
        uses: softprops/action-gh-release@v2
        with:
          files: |
            dist/**
```

### 🔐 Requires Secrets you set in GitHub {#🔐-需要你在-github-设置的-secrets}

Enter the repository → **Settings → Secrets → Actions → New repository secret**

| Secret name | value |
| ----------- | ---------------------------------------------------------------------------- |
| `NPM_TOKEN` | npm’s publish token (npmjs.com → Access Tokens → Generate Fine-grained Token) |

### 📌 Workflow triggering method {#📌-工作流触发方式}

Every time you push a **tag**:

```
git tag v1.0.0
git push origin v1.0.0
```

GitHub Actions automatically:

1. Install Node environment
2. `npm ci` install dependencies
3. Run test
4. `npm run build`
5. Publish to npm using `npm publish`
6. Create a GitHub Release with `dist/**` files
